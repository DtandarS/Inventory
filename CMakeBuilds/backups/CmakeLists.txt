cmake_minimum_required(VERSION 3.14.0)


# ---------------------------------- #


project(

Inventorio

)


# ---------------------------------- #


set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_NAME iOS)
set(DCMAKE_OSX_DEPLOYMENT_TARGET 26.0)


# ---------------------------------- #


add_subdirectory(

src/lib/SDL/
EXCLUDE_FROM_ALL

)


# ---------------------------------- #


add_subdirectory(

src/lib/SDL_image/
EXCLUDE_FROM_ALL

)


# ---------------------------------- #


add_executable(

main 

src/main.cpp

)


# ---------------------------------- #


target_include_directories(

main 

PRIVATE 

src/lib/SDL/include/SDL3/
src/lib/SDL_image/include/SDL3_image/

)


# ---------------------------------- #


target_link_libraries(

main 

PRIVATE 
SDL3::SDL3 
SDL3_image::SDL3_image 

)


# ---------------------------------- #
